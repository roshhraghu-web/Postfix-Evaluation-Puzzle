    class GradientPanel extends JPanel {

        Color c1;
        Color c2;

        GradientPanel(Color c1, Color c2) {

            this.c1 = c1;
            this.c2 = c2;
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            GradientPaint gradient =
                    new GradientPaint(
                            0,
                            0,
                            c1,
                            getWidth(),
                            getHeight(),
                            c2
                    );

            g2.setPaint(gradient);

            g2.fillRect(
                    0,
                    0,
                    getWidth(),
                    getHeight()
            );

            g2.dispose();
        }
    }


    // =========================================================
    // ROUNDED PANEL
    // =========================================================

    class RoundedPanel extends JPanel {

        private int cornerRadius = 35;
        private Color backgroundColor;

        RoundedPanel(Color backgroundColor) {

            this.backgroundColor = new Color(
            235,
            248,
            252
    );
;

            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            if (w <= 0 || h <= 0) {
                g2.dispose();
                return;
            }

            RoundRectangle2D roundedRectangle =
                    new RoundRectangle2D.Double(
                            0,
                            0,
                            w - 1,
                            h - 1,
                            cornerRadius,
                            cornerRadius
                    );

            g2.setColor(new Color(
                    235,
                    248,
                    252
            ));

            g2.fill(roundedRectangle);

            g2.dispose();

            super.paintComponent(g);
        }
    }

    // =========================================================
    // POSTFIX PUZZLE GENERATION
    // =========================================================

    void generatePuzzle() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        answered = false;

        puzzleNumber++;

        questionLabel.setText(
                "PUZZLE " +
                puzzleNumber +
                " / 3"
        );


        int a =
                random.nextInt(9) + 1;

        int b =
                random.nextInt(9) + 1;

        int c =
                random.nextInt(9) + 1;

        int op =
                random.nextInt(4);


        String postfixExpression;


        if (op == 0) {

            // a b + c *
            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " *";

        } else if (op == 1) {

            // a b * c +
            postfixExpression =
                    a + " " +
                    b + " * " +
                    c + " +";

        } else if (op == 2) {

            // a b - c *
            postfixExpression =
                    a + " " +
                    b + " - " +
                    c + " *";

        } else {

            // a b + c -
            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " -";
        }


        // Display postfix expression
        puzzle.setText(
                postfixExpression
        );


        // Calculate answer using STACK
        correctAnswer =
                evaluatePostfix(
                        postfixExpression
                );


        answer.setText(
                "ENTER YOUR ANSWER"
        );

        answer.setForeground(
                new Color(
                        130,
                        140,
                        155,
                        180
                )
        );

        result.setForeground(Color.BLACK);

        result.setText(" ");
    }


    // =========================================================
    // STACK-BASED POSTFIX EVALUATION
    // =========================================================

    int evaluatePostfix(String expression) {

        Stack<Integer> stack =
                new Stack<>();

        String[] tokens =
                expression.trim().split("\\s+");


        for (String token : tokens) {

            if (isNumber(token)) {

                stack.push(
                        Integer.parseInt(token)
                );

            } else {

                if (stack.size() < 2) {

                    throw new IllegalArgumentException(
                            "Invalid postfix expression"
                    );
                }

                int b = stack.pop();

                int a = stack.pop();

                int value;


                switch (token) {

                    case "+":
                        value = a + b;
                        break;

                    case "-":
                        value = a - b;
                        break;

                    case "*":
                        value = a * b;
                        break;

                    case "/":

                        if (b == 0) {

                            throw new ArithmeticException(
                                    "Division by zero"
                            );
                        }

                        value = a / b;
                        break;

                    default:

                        throw new IllegalArgumentException(
                                "Unknown operator: " +
                                token
                        );
                }


                stack.push(value);
            }
        }


        if (stack.size() != 1) {

            throw new IllegalArgumentException(
                    "Invalid postfix expression"
            );
        }


        return stack.pop();
    }


    // =========================================================
    // CHECK WHETHER TOKEN IS NUMBER
    // =========================================================

    boolean isNumber(String token) {

        try {

            Integer.parseInt(token);

            return true;

        } catch (NumberFormatException e) {

            return false;
        }
    }

    // =========================================================
    // AUTOMATIC ANSWER CHECK
    // =========================================================

    void setupAutoCheck() {

        answer.getDocument()
                .addDocumentListener(
                        new DocumentListener() {

                            public void insertUpdate(
                                    DocumentEvent e
                            ) {

                                startAutoCheck();
                            }


                            public void removeUpdate(
                                    DocumentEvent e
                            ) {

                                startAutoCheck();
                            }


                            public void changedUpdate(
                                    DocumentEvent e
                            ) {

                                startAutoCheck();
                            }
                        }
                );
    }


    void startAutoCheck() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;


        String value =
                answer.getText().trim();


        if (
                value.isEmpty() ||
                value.equals(
                        "ENTER YOUR ANSWER"
                )
        )
            return;


        autoCheckTimer =
                new javax.swing.Timer(
                        600,
                        e -> checkAnswer()
                );

        autoCheckTimer.setRepeats(false);

        autoCheckTimer.start();
    }


    // =========================================================
    // CHECK ANSWER
    // =========================================================

    void checkAnswer() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;


        try {

            String input =
                    answer.getText()
                            .trim();


            if (
                    input.equals(
                            "ENTER YOUR ANSWER"
                    ) ||
                    input.isEmpty()
            ) {

                result.setForeground(
                        Color.BLACK
                );

                result.setText(
                        "PLEASE ENTER A VALID NUMBER"
                );

                return;
            }


            int userAnswer =
                    Integer.parseInt(input);


            answered = true;


            if (
                    userAnswer ==
                    correctAnswer
            ) {

                score++;

                result.setForeground(
                        green
                );

                result.setText(
                        "✓ CORRECT!"
                );

            } else {

                result.setForeground(
                        red
                );

                result.setText(
                        "✗ WRONG! ANSWER: " +
                        correctAnswer
                );
            }


            scoreLabel.setText(
                    "SCORE: " +
                    score
            );


            leaderboard.put(
                    playerName,
                    score
            );

            saveLeaderboard();

            updateLeaderboard();


        } catch (Exception e) {

            result.setForeground(
                    Color.YELLOW
            );

            result.setText(
                    "PLEASE ENTER A VALID NUMBER"
            );
        }
    }

    // =========================================================
    // UPDATE LEADERBOARD
    // =========================================================

    void updateLeaderboard() {

        StringBuilder text =
                new StringBuilder();


        text.append(
                "PLAYER NAME              SCORE\n"
        );

        text.append(
                "--------------------------------\n"
        );


        int position = 1;


        for (
                Map.Entry<String, Integer> entry :
                leaderboard.entrySet()
        ) {

            text.append(
                    String.format(
                            "%-5d %-20s %d / 3\n",
                            position,
                            entry.getKey(),
                            entry.getValue()
                    )
            );

            position++;
        }


        if (leaderboard.isEmpty()) {

            text.append(
                    "\nNo players yet."
            );
        }


        leaderboardArea.setText(
                text.toString()
        );


        Dimension preferred =
                leaderboardArea.getPreferredSize();

        leaderboardArea.setSize(
                preferred
        );

        leaderboardArea.revalidate();

        leaderboardArea.repaint();
    }


    // =========================================================
    // SAVE LEADERBOARD
    // =========================================================

    void saveLeaderboard() {

        try {

            PrintWriter writer =
                    new PrintWriter(
                            new FileWriter(
                                    leaderboardFile
                            )
                    );


            for (
                    Map.Entry<String, Integer> entry :
                    leaderboard.entrySet()
            ) {

                writer.println(
                        entry.getKey() +
                        "|" +
                        entry.getValue()
                );
            }


            writer.close();


        } catch (IOException e) {

            JOptionPane.showMessageDialog(
                    this,
                    "Unable to save leaderboard."
            );
        }
    }


    // =========================================================
    // LOAD LEADERBOARD
    // =========================================================

    void loadLeaderboard() {

        File file =
                new File(
                        leaderboardFile
                );


        if (!file.exists()) {
            return;
        }


        try {

            BufferedReader reader =
                    new BufferedReader(
                            new FileReader(file)
                    );

            String line;


            while (
                    (line = reader.readLine())
                    != null
            ) {

                String[] data =
                        line.split("\\|");


                if (data.length == 2) {

                    String name =
                            data[0];

                    int savedScore =
                            Integer.parseInt(
                                    data[1]
                            );

                    leaderboard.put(
                            name,
                            savedScore
                    );
                }
            }


            reader.close();


        } catch (Exception e) {

            System.out.println(
                    "Unable to load leaderboard."
            );
        }
    }

    class GradientPanel extends JPanel {

        Color c1;
        Color c2;

        GradientPanel(Color c1, Color c2) {

            this.c1 = c1;
            this.c2 = c2;
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            GradientPaint gradient =
                    new GradientPaint(
                            0,
                            0,
                            c1,
                            getWidth(),
                            getHeight(),
                            c2
                    );

            g2.setPaint(gradient);

            g2.fillRect(
                    0,
                    0,
                    getWidth(),
                    getHeight()
            );

            g2.dispose();
        }
    }

    // =========================================================
    // ROUNDED PANEL
    // =========================================================

    class RoundedPanel extends JPanel {

        private int cornerRadius = 35;
        private Color backgroundColor;

        RoundedPanel(Color backgroundColor) {

            this.backgroundColor =
                    new Color(
                            235,
                            248,
                            252
                    );

            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            if (w <= 0 || h <= 0) {
                g2.dispose();
                return;
            }

            RoundRectangle2D roundedRectangle =
                    new RoundRectangle2D.Double(
                            0,
                            0,
                            w - 1,
                            h - 1,
                            cornerRadius,
                            cornerRadius
                    );

            g2.setColor(backgroundColor);

            g2.fill(roundedRectangle);

            g2.dispose();

            super.paintComponent(g);
        }
    }

    // =========================================================
    // ROUNDED TEXT FIELD

    class RoundedTextField extends JTextField {

        private int cornerRadius = 25;

        RoundedTextField() {

            setOpaque(false);

            setBorder(
                    BorderFactory.createEmptyBorder(
                            8,
                            18,
                            8,
                            18
                    )
            );
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            if (w <= 0 || h <= 0) {
                g2.dispose();
                return;
            }

            // LIGHT CYAN-BLUE INPUT BOX
            g2.setColor(
                    new Color(
                            210,
                            245,
                            250
                    )
            );

            RoundRectangle2D rounded =
                    new RoundRectangle2D.Double(
                            0,
                            0,
                            w - 1,
                            h - 1,
                            cornerRadius,
                            cornerRadius
                    );

            g2.fill(rounded);

            // Keep text inside rounded shape
            g2.clip(rounded);

            super.paintComponent(g2);

            g2.dispose();
        }

        @Override
        protected void paintBorder(Graphics g) {
            // NO BORDER
        }
    }

    // =========================================================
    // POSTFIX PUZZLE GENERATION
    // =========================================================

    void generatePuzzle() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        answered = false;

        puzzleNumber++;

        questionLabel.setText(
                "PUZZLE " +
                puzzleNumber +
                " / 3"
        );

        int a =
                random.nextInt(9) + 1;

        int b =
                random.nextInt(9) + 1;

        int c =
                random.nextInt(9) + 1;

        int op =
                random.nextInt(4);

        String postfixExpression;

        if (op == 0) {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " *";

        } else if (op == 1) {

            postfixExpression =
                    a + " " +
                    b + " * " +
                    c + " +";

        } else if (op == 2) {

            postfixExpression =
                    a + " " +
                    b + " - " +
                    c + " *";

        } else {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " -";
        }

        puzzle.setText(
                postfixExpression
        );

        correctAnswer =
                evaluatePostfix(
                        postfixExpression
                );

        answer.setText(
                "ENTER YOUR ANSWER"
        );

        answer.setForeground(
                new Color(
                        70,
                        190,
                        220,
                        220
                )
        );

        result.setForeground(Color.BLACK);

        result.setText(" ");
    }

    // =========================================================
    // STACK-BASED POSTFIX EVALUATION
    // =========================================================

    int evaluatePostfix(String expression) {

        Stack<Integer> stack =
                new Stack<>();

        String[] tokens =
                expression.trim().split("\\s+");

        for (String token : tokens) {

            if (isNumber(token)) {

                stack.push(
                        Integer.parseInt(token)
                );

            } else {

                if (stack.size() < 2) {

                    throw new IllegalArgumentException(
                            "Invalid postfix expression"
                    );
                }

                int b = stack.pop();

                int a = stack.pop();

                int value;

                switch (token) {

                    case "+":
                        value = a + b;
                        break;

                    case "-":
                        value = a - b;
                        break;

                    case "*":
                        value = a * b;
                        break;

                    case "/":

                        if (b == 0) {

                            throw new ArithmeticException(
                                    "Division by zero"
                            );
                        }

                        value = a / b;
                        break;

                    default:

                        throw new IllegalArgumentException(
                                "Unknown operator: " +
                                token
                        );
                }

                stack.push(value);
            }
        }

        if (stack.size() != 1) {

            throw new IllegalArgumentException(
                    "Invalid postfix expression"
            );
        }

        return stack.pop();
    }

    // =========================================================
    // CHECK WHETHER TOKEN IS NUMBER
    // =========================================================

    boolean isNumber(String token) {

        try {

            Integer.parseInt(token);

            return true;

        } catch (NumberFormatException e) {

            return false;
        }
    }

    void startAutoCheck() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        String value =
                answer.getText().trim();

        if (
                value.isEmpty() ||
                value.equals(
                        "ENTER YOUR ANSWER"
                )
        )
            return;

        autoCheckTimer =
                new javax.swing.Timer(
                        600,
                        e -> checkAnswer()
                );

        autoCheckTimer.setRepeats(false);

        autoCheckTimer.start();
    }

    // =========================================================
    // CHECK ANSWER
    // =========================================================

    void checkAnswer() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        try {

            String input =
                    answer.getText()
                            .trim();

            if (
                    input.equals(
                            "ENTER YOUR ANSWER"
                    ) ||
                    input.isEmpty()
            ) {

                result.setForeground(
                        Color.BLACK
                );

                result.setText(
                        "PLEASE ENTER A VALID NUMBER"
                );

                return;
            }

            int userAnswer =
                    Integer.parseInt(input);

            answered = true;

            if (
                    userAnswer ==
                    correctAnswer
            ) {

                score++;

                result.setForeground(
                        green
                );

                result.setText(
                        "✓ CORRECT!"
                );

            } else {

                result.setForeground(
                        red
                );

                result.setText(
                        "✗ WRONG! ANSWER: " +
                        correctAnswer
                );
            }

            scoreLabel.setText(
                    "SCORE: " +
                    score
            );

            leaderboard.put(
                    playerName,
                    score
            );

            saveLeaderboard();

            updateLeaderboard();

        } catch (Exception e) {

            result.setForeground(
                    Color.YELLOW
            );

            result.setText(
                    "PLEASE ENTER A VALID NUMBER"
            );
        }
    }

    // =========================================================
    // UPDATE LEADERBOARD
    // =========================================================

    void updateLeaderboard() {

        StringBuilder text =
                new StringBuilder();

        text.append(
                "PLAYER NAME              SCORE\n"
        );

        text.append(
                "--------------------------------\n"
        );

        int position = 1;

        for (
                Map.Entry<String, Integer> entry :
                leaderboard.entrySet()
        ) {

            text.append(
                    String.format(
                            "%-5d %-20s %d / 3\n",
                            position,
                            entry.getKey(),
                            entry.getValue()
                    )
            );

            position++;
        }

        if (leaderboard.isEmpty()) {

            text.append(
                    "\nNo players yet."
            );
        }

        leaderboardArea.setText(
                text.toString()
        );

        Dimension preferred =
                leaderboardArea.getPreferredSize();

        leaderboardArea.setSize(
                preferred
        );

        leaderboardArea.revalidate();

        leaderboardArea.repaint();
    }

    // =========================================================
    // SAVE LEADERBOARD
    // =========================================================

    void saveLeaderboard() {

        try {

            PrintWriter writer =
                    new PrintWriter(
                            new FileWriter(
                                    leaderboardFile
                            )
                    );

            for (
                    Map.Entry<String, Integer> entry :
                    leaderboard.entrySet()
            ) {

                writer.println(
                        entry.getKey() +
                        "|" +
                        entry.getValue()
                );
            }

            writer.close();

        } catch (IOException e) {

            JOptionPane.showMessageDialog(
                    this,
                    "Unable to save leaderboard."
            );
        }
    }

    // =========================================================
    // LOAD LEADERBOARD
    // =========================================================

    void loadLeaderboard() {

        File file =
                new File(
                        leaderboardFile
                );

        if (!file.exists()) {
            return;
        }

        try {

            BufferedReader reader =
                    new BufferedReader(
                            new FileReader(file)
                    );

            String line;

            while (
                    (line = reader.readLine())
                    != null
            ) {

                String[] data =
                        line.split("\\|");

                if (data.length == 2) {

                    String name =
                            data[0];

                    int savedScore =
                            Integer.parseInt(
                                    data[1]
                            );

                    leaderboard.put(
                            name,
                            savedScore
                    );
                }
            }

            reader.close();

        } catch (Exception e) {

            System.out.println(
                    "Unable to load leaderboard."
            );
        }
    }

    class GradientPanel extends JPanel {

        Color c1;
        Color c2;

        GradientPanel(Color c1, Color c2) {

            this.c1 = c1;
            this.c2 = c2;
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            GradientPaint gradient =
                    new GradientPaint(
                            0,
                            0,
                            c1,
                            getWidth(),
                            getHeight(),
                            c2
                    );

            g2.setPaint(gradient);

            g2.fillRect(
                    0,
                    0,
                    getWidth(),
                    getHeight()
            );

            g2.dispose();
        }
    }

    // =========================================================
    // ROUNDED PANEL
    // =========================================================

    class RoundedPanel extends JPanel {

        private int cornerRadius = 35;
        private Color backgroundColor;

        RoundedPanel(Color backgroundColor) {

            this.backgroundColor =
                    new Color(
                            135,
                            205,
                            236
                    );
        //this.backgroundColor = backgroundColor;

            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            if (w <= 0 || h <= 0) {
                g2.dispose();
                return;
            }

            RoundRectangle2D roundedRectangle =
                    new RoundRectangle2D.Double(
                            0,
                            0,
                            w - 1,
                            h - 1,
                            cornerRadius,
                            cornerRadius
                    );

            g2.setColor(backgroundColor);

            g2.fill(roundedRectangle);

            g2.dispose();

            super.paintComponent(g);
        }
    }

    // =========================================================
    // POSTFIX PUZZLE GENERATION
    // =========================================================

    void generatePuzzle() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        answered = false;

        puzzleNumber++;

        questionLabel.setText(
                "PUZZLE " +
                puzzleNumber +
                " / 3"
        );

        int a =
                random.nextInt(9) + 1;

        int b =
                random.nextInt(9) + 1;

        int c =
                random.nextInt(9) + 1;

        int op =
                random.nextInt(4);

        String postfixExpression;

        if (op == 0) {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " *";

        } else if (op == 1) {

            postfixExpression =
                    a + " " +
                    b + " * " +
                    c + " +";

        } else if (op == 2) {

            postfixExpression =
                    a + " " +
                    b + " - " +
                    c + " *";

        } else {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " -";
        }

        puzzle.setText(
                postfixExpression
        );

        correctAnswer =
                evaluatePostfix(
                        postfixExpression
                );

        answer.setText(
                "ENTER YOUR ANSWER"
        );

        answer.setForeground(
                new Color(
                        70,
                        190,
                        220,
                        220
                )
        );

        result.setForeground(Color.BLACK);

        result.setText(" ");
    }

    // =========================================================
    // STACK-BASED POSTFIX EVALUATION
    // =========================================================

    int evaluatePostfix(String expression) {

        Stack<Integer> stack =
                new Stack<>();

        String[] tokens =
                expression.trim().split("\\s+");

        for (String token : tokens) {

            if (isNumber(token)) {

                stack.push(
                        Integer.parseInt(token)
                );

            } else {

                if (stack.size() < 2) {

                    throw new IllegalArgumentException(
                            "Invalid postfix expression"
                    );
                }

                int b = stack.pop();

                int a = stack.pop();

                int value;

                switch (token) {

                    case "+":
                        value = a + b;
                        break;

                    case "-":
                        value = a - b;
                        break;

                    case "*":
                        value = a * b;
                        break;

                    case "/":

                        if (b == 0) {

                            throw new ArithmeticException(
                                    "Division by zero"
                            );
                        }

                        value = a / b;
                        break;

                    default:

                        throw new IllegalArgumentException(
                                "Unknown operator: " +
                                token
                        );
                }

                stack.push(value);
            }
        }

        if (stack.size() != 1) {

            throw new IllegalArgumentException(
                    "Invalid postfix expression"
            );
        }

        return stack.pop();
    }

    // =========================================================
    // CHECK WHETHER TOKEN IS NUMBER
    // =========================================================

    boolean isNumber(String token) {

        try {

            Integer.parseInt(token);

            return true;

        } catch (NumberFormatException e) {

            return false;
        }
    }

    void startAutoCheck() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        String value =
                answer.getText().trim();

        if (
                value.isEmpty() ||
                value.equals(
                        "ENTER YOUR ANSWER"
                )
        )
            return;

        autoCheckTimer =
                new javax.swing.Timer(
                        600,
                        e -> checkAnswer()
                );

        autoCheckTimer.setRepeats(false);

        autoCheckTimer.start();
    }

    // =========================================================
    // CHECK ANSWER
    // =========================================================

    void checkAnswer() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        try {

            String input =
                    answer.getText()
                            .trim();

            if (
                    input.equals(
                            "ENTER YOUR ANSWER"
                    ) ||
                    input.isEmpty()
            ) {

                result.setForeground(
                        Color.BLACK
                );

                result.setText(
                        "PLEASE ENTER A VALID NUMBER"
                );

                return;
            }

            int userAnswer =
                    Integer.parseInt(input);

            answered = true;

            if (
                    userAnswer ==
                    correctAnswer
            ) {

                score++;

                result.setForeground(
                        green
                );

                result.setText(
                        "✓ CORRECT!"
                );

            } else {

                result.setForeground(
                        red
                );

                result.setText(
                        "✗ WRONG! ANSWER: " +
                        correctAnswer
                );
            }

            scoreLabel.setText(
                    "SCORE: " +
                    score
            );

            leaderboard.put(
                    playerName,
                    score
            );

            saveLeaderboard();

            updateLeaderboard();

        } catch (Exception e) {

            result.setForeground(
                    Color.YELLOW
            );

            result.setText(
                    "PLEASE ENTER A VALID NUMBER"
            );
        }
    }

    // =========================================================

    void updateLeaderboard() {

        StringBuilder text =
                new StringBuilder();

        text.append(
                "PLAYER NAME              SCORE\n"
        );

        text.append(
                "--------------------------------\n"
        );

        int position = 1;

        for (
                Map.Entry<String, Integer> entry :
                leaderboard.entrySet()
        ) {

            text.append(
                    String.format(
                            "%-5d %-20s %d / 3\n",
                            position,
                            entry.getKey(),
                            entry.getValue()
                    )
            );

            position++;
        }

        if (leaderboard.isEmpty()) {

            text.append(
                    "\nNo players yet."
            );
        }

        leaderboardArea.setText(
                text.toString()
        );

        Dimension preferred =
                leaderboardArea.getPreferredSize();

        leaderboardArea.setSize(
                preferred
        );

        leaderboardArea.revalidate();

        leaderboardArea.repaint();
    }

    // =========================================================
    // SAVE LEADERBOARD
    // =========================================================

    void saveLeaderboard() {

        try {

            PrintWriter writer =
                    new PrintWriter(
                            new FileWriter(
                                    leaderboardFile
                            )
                    );

            for (
                    Map.Entry<String, Integer> entry :
                    leaderboard.entrySet()
            ) {

                writer.println(
                        entry.getKey() +
                        "|" +
                        entry.getValue()
                );
            }

            writer.close();

        } catch (IOException e) {

            JOptionPane.showMessageDialog(
                    this,
                    "Unable to save leaderboard."
            );
        }
    }

    // =========================================================
    // LOAD LEADERBOARD
    // =========================================================

    void loadLeaderboard() {

        File file =
                new File(
                        leaderboardFile
                );

        if (!file.exists()) {
            return;
        }

        try {

            BufferedReader reader =
                    new BufferedReader(
                            new FileReader(file)
                    );

            String line;

            while (
                    (line = reader.readLine())
                    != null
            ) {

                String[] data =
                        line.split("\\|");

                if (data.length == 2) {

                    String name =
                            data[0];

                    int savedScore =
                            Integer.parseInt(
                                    data[1]
                            );

                    leaderboard.put(
                            name,
                            savedScore
                    );
                }
            }

            reader.close();

        } catch (Exception e) {

            System.out.println(
                    "Unable to load leaderboard."
            );
        }
    }

    class GradientPanel extends JPanel {

        Color c1;
        Color c2;

        GradientPanel(Color c1, Color c2) {

            this.c1 = c1;
            this.c2 = c2;
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            GradientPaint gradient =
                    new GradientPaint(
                            0,
                            0,
                            c1,
                            getWidth(),
                            getHeight(),
                            c2
                    );

            g2.setPaint(gradient);

            g2.fillRect(
                    0,
                    0,
                    getWidth(),
                    getHeight()
            );

            g2.dispose();
        }
    }

    class RoundedPanel extends JPanel {

        private int cornerRadius = 35;
        private Color backgroundColor;

        RoundedPanel(Color backgroundColor) {

            this.backgroundColor =
                    new Color(
                            135,
                            205,
                            236
                    );

            // this.backgroundColor = backgroundColor;

            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 =
                    (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int w = getWidth();
            int h = getHeight();

            if (w <= 0 || h <= 0) {
                g2.dispose();
                return;
            }

            RoundRectangle2D roundedRectangle =
                    new RoundRectangle2D.Double(
                            0,
                            0,
                            w - 1,
                            h - 1,
                            cornerRadius,
                            cornerRadius
                    );

            g2.setColor(backgroundColor);

            g2.fill(roundedRectangle);

            g2.dispose();

            super.paintComponent(g);
        }
    }

    // =========================================================
    // POSTFIX PUZZLE GENERATION
    // =========================================================

    void generatePuzzle() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        answered = false;

        puzzleNumber++;

        questionLabel.setText(
                "PUZZLE " +
                puzzleNumber +
                " / 3"
        );

        int a =
                random.nextInt(9) + 1;

        int b =
                random.nextInt(9) + 1;

        int c =
                random.nextInt(9) + 1;

        int op =
                random.nextInt(4);

        String postfixExpression;

        if (op == 0) {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " *";

        } else if (op == 1) {

            postfixExpression =
                    a + " " +
                    b + " * " +
                    c + " +";

        } else if (op == 2) {

            postfixExpression =
                    a + " " +
                    b + " - " +
                    c + " *";

        } else {

            postfixExpression =
                    a + " " +
                    b + " + " +
                    c + " -";
        }

        puzzle.setText(
                postfixExpression
        );

        correctAnswer =
                evaluatePostfix(
                        postfixExpression
                );

        answer.setText(
                "ENTER YOUR ANSWER"
        );

        answer.setForeground(
                new Color(
                        70,
                        190,
                        220,
                        220
                )
        );

        result.setForeground(Color.BLACK);

        result.setText(" ");
    }

    // =========================================================
    // STACK-BASED POSTFIX EVALUATION
    // =========================================================

    int evaluatePostfix(String expression) {

        Stack<Integer> stack =
                new Stack<>();

        String[] tokens =
                expression.trim().split("\\s+");

        for (String token : tokens) {

            if (isNumber(token)) {

                stack.push(
                        Integer.parseInt(token)
                );

            } else {

                if (stack.size() < 2) {

                    throw new IllegalArgumentException(
                            "Invalid postfix expression"
                    );
                }

                int b = stack.pop();

                int a = stack.pop();

                int value;

                switch (token) {

                    case "+":
                        value = a + b;
                        break;

                    case "-":
                        value = a - b;
                        break;

                    case "*":
                        value = a * b;
                        break;

                    case "/":

                        if (b == 0) {

                            throw new ArithmeticException(
                                    "Division by zero"
                            );
                        }

                        value = a / b;
                        break;

                    default:

                        throw new IllegalArgumentException(
                                "Unknown operator: " +
                                token
                        );
                }

                stack.push(value);
            }
        }

        if (stack.size() != 1) {

            throw new IllegalArgumentException(
                    "Invalid postfix expression"
            );
        }

        return stack.pop();
    }

    // =========================================================
    // CHECK WHETHER TOKEN IS NUMBER
    // =========================================================

    boolean isNumber(String token) {

        try {

            Integer.parseInt(token);

            return true;

        } catch (NumberFormatException e) {

            return false;
        }
    }

    void startAutoCheck() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        String value =
                answer.getText().trim();

        if (
                value.isEmpty() ||
                value.equals(
                        "ENTER YOUR ANSWER"
                )
        )
            return;

        autoCheckTimer =
                new javax.swing.Timer(
                        600,
                        e -> checkAnswer()
                );

        autoCheckTimer.setRepeats(false);

        autoCheckTimer.start();
    }

    // =========================================================
    // CHECK ANSWER
    // =========================================================

    void checkAnswer() {

        if (autoCheckTimer != null)
            autoCheckTimer.stop();

        if (answered)
            return;

        try {

            String input =
                    answer.getText()
                            .trim();

            if (
                    input.equals(
                            "ENTER YOUR ANSWER"
                    ) ||
                    input.isEmpty()
            ) {

                result.setForeground(
                        Color.BLACK
                );

                result.setText(
                        "PLEASE ENTER A VALID NUMBER"
                );

                return;
            }

            int userAnswer =
                    Integer.parseInt(input);

            answered = true;

            if (
                    userAnswer ==
                    correctAnswer
            ) {

                score++;

                result.setForeground(
                        green
                );

                result.setText(
                        "✓ CORRECT!"
                );

            } else {

                result.setForeground(
                        red
                );

                result.setText(
                        "✗ WRONG! ANSWER: " +
                        correctAnswer
                );
            }

            scoreLabel.setText(
                    "SCORE: " +
                    score
            );

            leaderboard.put(
                    playerName,
                    score
            );

            saveLeaderboard();

            updateLeaderboard();

        } catch (Exception e) {

            result.setForeground(
                    Color.YELLOW
            );

            result.setText(
                    "PLEASE ENTER A VALID NUMBER"
            );
        }
    }

    // =========================================================
    // UPDATE LEADERBOARD
    // =========================================================

    void updateLeaderboard() {

        StringBuilder text =
                new StringBuilder();

        text.append(
                "PLAYER NAME              SCORE\n"
        );

        text.append(
                "--------------------------------\n"
        );

        int position = 1;

        for (
                Map.Entry<String, Integer> entry :
                leaderboard.entrySet()
        ) {

            text.append(
                    String.format(
                            "%-5d %-20s %d / 3\n",
                            position,
                            entry.getKey(),
                            entry.getValue()
                    )
            );

            position++;
        }

        if (leaderboard.isEmpty()) {

            text.append(
                    "\nNo players yet."
            );
        }

        leaderboardArea.setText(
                text.toString()
        );

        Dimension preferred =
                leaderboardArea.getPreferredSize();

        leaderboardArea.setSize(
                preferred
        );

        leaderboardArea.revalidate();

        leaderboardArea.repaint();
    }

    // =========================================================
    // SAVE LEADERBOARD
    // =========================================================

    void saveLeaderboard() {

        try {

            PrintWriter writer =
                    new PrintWriter(
                            new FileWriter(
                                    leaderboardFile
                            )
                    );

            for (
                    Map.Entry<String, Integer> entry :
                    leaderboard.entrySet()
            ) {

                writer.println(
                        entry.getKey() +
                        "|" +
                        entry.getValue()
                );
            }

            writer.close();

        } catch (IOException e) {

            JOptionPane.showMessageDialog(
                    this,
                    "Unable to save leaderboard."
            );
        }
    }

    // =========================================================
    // LOAD LEADERBOARD
    // =========================================================

    void loadLeaderboard() {

        File file =
                new File(
                        leaderboardFile
                );

        if (!file.exists()) {
            return;
        }

        try {

            BufferedReader reader =
                    new BufferedReader(
                            new FileReader(file)
                    );

            String line;

            while (
                    (line = reader.readLine())
                    != null
            ) {

                String[] data =
                        line.split("\\|");

                if (data.length == 2) {

                    String name =
                            data[0];

                    int savedScore =
                            Integer.parseInt(
                                    data[1]
                            );

                    leaderboard.put(
                            name,
                            savedScore
                    );
                }
            }

            reader.close();

        } catch (Exception e) {

            System.out.println(
                    "Unable to load leaderboard."
            );
        }
    }
